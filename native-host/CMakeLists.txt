cmake_minimum_required(VERSION 3.22)
project(alloy2581_host LANGUAGES CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "The runtime host requires the Android NDK")
endif()
if(NOT EXISTS "${ALLOY_ENGINE_LIBRARY}" OR NOT EXISTS "${ALLOY_ENGINE_HEADERS}/platform/CCApplication.h")
    message(FATAL_ERROR "Verified 2581 engine library and source headers are required")
endif()

add_library(cocos2djs SHARED IMPORTED)
set_target_properties(cocos2djs PROPERTIES IMPORTED_LOCATION "${ALLOY_ENGINE_LIBRARY}")
add_library(alloy2581 SHARED runtime.cpp)
target_compile_features(alloy2581 PRIVATE cxx_std_17)
target_compile_definitions(alloy2581 PRIVATE ANDROID=1)
target_include_directories(alloy2581 SYSTEM PRIVATE "${ALLOY_ENGINE_HEADERS}")
target_compile_options(alloy2581 PRIVATE -Wall -Wextra -Werror)
target_link_options(alloy2581 PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
target_link_libraries(alloy2581 PRIVATE cocos2djs android log GLESv2 dl)
